#include <bits/stdc++.h>
using namespace std;

int main() {
    int n,q;
    cin>>n>>q;
    long long a[n+1];
    a[0]=0;
    for(int i=1; i<=n; i++){
        cin>>a[i];
        a[i]=a[i-1]^a[i];
    }
    while(q--){
        int l,r;
        cin>>l>>r;
        long long ans=a[r]^a[l-1];
        cout<<ans<<"\n";
    }
    return 0;
}
